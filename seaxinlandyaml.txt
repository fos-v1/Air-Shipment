/shipment/sea:
    post:
      tags: [Shipment Sharing]
      summary: Create a new sea shipment
      description: >
        Creates a sea shipment in Freightoscope.
        Authentication is provided via query parameters in the request URL.
      operationId: createSeaShipment
#      parameters:
#        - in: query
#          name: UserName
#          required: true
#          schema:
#            type: string
#          description: Application user name.
#          example: TEST_USERNAME
#         - in: query
#          name: Password
#         required: true
#          schema:
#            type: string
#            format: password
#          description: Application password.
#         example: TEST_PASSWORD
#        - in: query
#          name: CompanyId
#          required: true
#          schema:
#            type: integer
#          description: Numeric company identifier.
#          example: 20263
      requestBody:
        required: true
        content:
          application/json:
            schema:
              type: object
              required:
                - UserName
                - Password
                - CompanyId
                - ShipmentData
              properties:
                UserName:
                  type: string
                  example: TEST_USERNAME      
                  description: UserName as configured in the corresponding branch in Freightoscope.
                Password:
                  type: string  
                  example: TEST_PASSWORD
                  description: Password as configured in the corresponding branch in Freightoscope.
                CompanyId:
                  type: integer
                  example: 20263     
                  description: Numeric company identifier of the corresponding branch in Freightoscope.
                ShipmentData:
                  $ref: '#/components/schemas/Sea Shipment'
            
      responses:
        '200':
          description: Shipment created Successfully
          content:
            application/json:
              schema:
                type: object
                properties:
                  Message:
                    type: string
                    example: Shipment created Successfully
                  Code:
                    type: integer
                    example: 200
                required:
                  - Message
                  - Code
        '400':
          description: Shipment creation failed
          content:
            application/json:
              schema:
                type: object
                properties:
                  Message:
                    type: string
                    example: Shipment creation failed
                  Code:
                    type: integer
                    example: 400
                required:
                  - Message
                  - Code
        '403':
          description: Your company does not have permission to access shared shipments
          content:
            application/json:
              schema:
                type: object
                properties:
                  Message:
                    type: string
                    example: Your company does not have permission to access shared shipments
                  Code:
                    type: integer
                    example: 403
                required:
                  - Message
                  - Code      
        '500':

                  Message:
                    type: string
